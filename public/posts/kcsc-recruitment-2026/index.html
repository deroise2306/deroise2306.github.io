Nhìn chung thì một bài RSA rất căn bản với các lỗi hoặc yêu cầu các kĩ thuật exploit cơ bản của RSA. Lần lượt từng phase của challenge thì sẽ như sau:

  • Phase 1: Small Public Exponent (small e) với e1 = 7.
  • Phase 2: một bộ các giá trị n khác nhau, các giá trị ct khác nhau tương ứng nhưng chung một giá trị e => Hastad Broadcast => sử dụng CRT để giải.
  • Phase 3: đề cho thêm một cặp E và d => có thể từ đó tìm được giá trị $\varphi(n)$ để giải mã.
  • Phase 4: mặc dù đi qua hàm băm SHA256, nhưng với 7 kí tự còn lại thì việc brute là hoàn toàn có thể.

def recover_small_e_linear(c, n, byte_len):
    min_msg = bytes_to_long(b'0' * byte_len)
    max_msg = bytes_to_long(b'f' * byte_len)
    k_min = (pow(min_msg, 7) - c) // n
    k_max = (pow(max_msg, 7) - c) // n

    
    for k in range(k_min, k_max + 2):
        val = c + k * n
        m, exact = gmpy2.iroot(val, 7)
        if exact:
            cand = long_to_bytes(m)
            if len(cand) == byte_len and all(chr(x) in "0123456789abcdef" for x in cand):
                return cand
    return None

# my prime generator
def get_prime(factor_bits, nbits):
    while True:
        p = 2
        while p.bit_length() + factor_bits < nbits:
            p *= getPrime(factor_bits)
        current_len = p.bit_length()
        needed_bits = nbits - current_len
        
        if needed_bits < 2:
            continue
        try:
            last_prime = getPrime(needed_bits)
        except ValueError:
            continue
        
        p = p * last_prime + 1
        if p.bit_length() == nbits and isPrime(p):
            return p

def solve_linear_system_mod(prime, ciphertexts, exponent_matrix):
    F = GF(prime)
    g = F.multiplicative_generator()
    order = prime - 1
    
    print(f"    [+] Log base: {g}")

    try:
        Y = vector(Zmod(order), [F(c).log(g) for c in ciphertexts])
    except ValueError as e:
        print(f"    [-] Error computing log: {e}")
        return []
    
    M = matrix(Zmod(order), exponent_matrix)
    
    try:
        x_particular = M.solve_right(Y)
    except ValueError:
        print(f"    [-] No solution found.")
        return []

    K = M.right_kernel()
    gens = K.gens() 
    
    gen_orders = []

        v_list = [lift(x) for x in vec]
        common = gcd(v_list + [order])
        gen_orders.append(order // common)

        x_sol = x_particular + offset
